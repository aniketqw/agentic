import math

def probabilities(p):
    """Reject an invalid categorical distribution before filtering/scoring."""
    p=np.asarray(p,dtype=float)
    if p.ndim!=1 or not len(p) or not np.all(np.isfinite(p)) or np.any(p<0) or not np.isclose(p.sum(),1,atol=1e-10):
        raise ValueError('nonempty nonnegative normalized finite probabilities required')
    return p


def filter_distribution(p, parameter, kind):
    """Min-p relative cutoff or locally typical cumulative mass, then normalize."""
    p=probabilities(p)
    if not 0<parameter<=1:raise ValueError('parameter in (0,1] required')
    if kind=='min_p':keep=np.flatnonzero(p>=parameter*p.max()).tolist()
    elif kind=='typical':
        entropy=-sum(x*math.log(x) for x in p if x)
        order=sorted(range(len(p)),key=lambda i:(abs(-math.log(p[i])-entropy) if p[i] else math.inf,i))
        keep=[];mass=0
        for i in order:
            if not p[i]:break
            keep.append(i);mass+=p[i]
            if mass>=parameter:break
    else:raise ValueError('unknown filter')
    out=np.zeros_like(p);out[keep]=p[keep]/p[keep].sum()
    return keep,out


def repetition_penalty(logits, seen, penalty):
    """Suppress positive/negative logits consistently; process each seen ID once."""
    if not math.isfinite(penalty) or penalty<=0:raise ValueError('positive finite penalty required')
    z=np.array(logits,dtype=float,copy=True)
    for i in set(seen):
        if i<0 or i>=len(z):raise ValueError('seen token outside vocabulary')
        z[i]=z[i]/penalty if z[i]>=0 else z[i]*penalty
    return z


def contrastive_candidate(probability, hidden, context, alpha=.5):
    """Compute actual cosine similarities and candidate tradeoff, not supplied scores."""
    p=probabilities(probability);h=np.asarray(hidden,float);c=np.asarray(context,float)
    if not 0<=alpha<=1 or h.ndim!=2 or c.ndim!=2 or len(h)!=len(p) or h.shape[1]!=c.shape[1] or not len(c):raise ValueError('incompatible inputs')
    hn=np.linalg.norm(h,axis=1);cn=np.linalg.norm(c,axis=1)
    if np.any(hn==0) or np.any(cn==0):raise ValueError('cosine needs nonzero vectors')
    similarity=(h @ c.T)/(hn[:,None]*cn[None,:])
    scores=(1-alpha)*p-alpha*similarity.max(axis=1)
    return int(np.argmax(scores)),scores,similarity


def float16_decode(bits):
    """Decode all binary16 patterns, including subnormals, signed zero and NaNs."""
    if not isinstance(bits,int) or not 0<=bits<65536:raise ValueError('16-bit integer required')
    sign=-1 if bits>>15 else 1;e=(bits>>10)&31;m=bits&1023
    if e==0:return math.copysign(m*2**-24,sign)
    if e==31:return math.copysign(math.inf,sign) if m==0 else math.nan
    return sign*(1+m/1024)*2**(e-15)


def quantized_dot(x_codes,w_codes,x_step,w_step,output_step=None):
    """Keep dequantization steps in real-units/code; accumulate integers exactly."""
    if min(x_step,w_step)<=0 or (output_step is not None and output_step<=0):raise ValueError('positive steps required')
    x=np.asarray(x_codes);w=np.asarray(w_codes)
    if x.ndim!=1 or w.shape!=x.shape or not np.issubdtype(x.dtype,np.integer) or not np.issubdtype(w.dtype,np.integer):raise ValueError('matching integer vectors required')
    accumulator=sum(int(a)*int(b) for a,b in zip(x,w))
    real=x_step*w_step*accumulator
    return accumulator,real,(round(real/output_step) if output_step is not None else None)


def compensate_fixed_coordinate(hessian,error_index,fixed_error):
    """Solve the exact quadratic optimum in remaining coordinates (one GPTQ idea)."""
    h=np.asarray(hessian,float);n=len(h)
    if h.shape!=(n,n) or not np.allclose(h,h.T) or np.min(np.linalg.eigvalsh(h))<=0:raise ValueError('positive-definite symmetric H required')
    rest=[i for i in range(n) if i!=error_index]
    error=np.zeros(n);error[error_index]=fixed_error
    error[rest]=np.linalg.solve(h[np.ix_(rest,rest)],-h[rest,error_index]*fixed_error)
    return error,float(error @ h @ error)


def two_token_distribution(theta):
    """Three Bernoulli logits: root and the two first-token-conditioned branches."""
    theta=np.asarray(theta,float)
    if theta.shape!=(3,) or not np.all(np.isfinite(theta)):raise ValueError('three finite logits required')
    sigmoid=lambda t:1/(1+math.exp(-float(t)))
    r,left,right=map(sigmoid,theta)
    return np.array([(1-r)*(1-left),(1-r)*left,r*(1-right),r*right])


def sequence_reverse_kl(theta,teacher):
    """Enumerate all two-token paths and full score-function gradients exactly."""
    p=probabilities(teacher);q=two_token_distribution(theta)
    if p.shape!=(4,) or np.any(p==0):raise ValueError('positive teacher on four paths required')
    r,left,right=[1/(1+math.exp(-float(t))) for t in theta]
    ratios=np.log(q/p);loss=float(q @ ratios);gradient=np.zeros(3)
    for index,(first,second) in enumerate([(0,0),(0,1),(1,0),(1,1)]):
        score=np.array([first-r,second-left if first==0 else 0,second-right if first==1 else 0])
        gradient+=q[index]*(ratios[index]+1)*score
    return loss,gradient,q


def importance_expectation(target,proposal,values):
    """Exact finite proposal weighting; require coverage wherever target is positive."""
    q=probabilities(target);m=probabilities(proposal);v=np.asarray(values,float)
    if q.shape!=m.shape or v.shape!=q.shape or np.any((q>0)&(m==0)):raise ValueError('shape/support mismatch')
    w=np.divide(q,m,out=np.zeros_like(q),where=m>0)
    return float(np.sum(m*w*v)),w


def expected_k3(student,reference):
    """k3 value equality holds under student sampling; gradients need separate care."""
    q=probabilities(student);p=probabilities(reference)
    if q.shape!=p.shape or np.any(q==0) or np.any(p==0):raise ValueError('positive common support required')

def ece(confidences,correct,boundaries=(0,.5,1)):
    """Weighted calibration error; bins are [low,high), final bin includes 1."""
    conf=np.asarray(confidences,float);corr=np.asarray(correct,bool)
    if conf.shape!=corr.shape or conf.ndim!=1 or not len(conf) or np.any((conf<0)|(conf>1)):raise ValueError('valid paired predictions required')
    if list(boundaries)!=sorted(boundaries) or boundaries[0]!=0 or boundaries[-1]!=1:raise ValueError('bins must partition [0,1]')
    total=0;parts=[]
    for i,(lo,hi) in enumerate(zip(boundaries[:-1],boundaries[1:])):
        take=(conf>=lo)&(conf<=hi if i==len(boundaries)-2 else conf<hi);n=int(take.sum())
        if n:

            parts.append({'count':n,'accuracy':float(corr[take].mean()),'confidence':float(conf[take].mean()),'gap':gap})
    return float(total),parts


def distinct_n(responses,n=4):

    if n<1:raise ValueError('n must be positive')
    grams=[tuple(words[i:i+n]) for words in responses for i in range(max(0,len(words)-n+1))]
    return (len(set(grams))/len(grams) if grams else None),len(grams)

"""Executable NumPy teaching companion; no pretrained model or distributed benchmark.

cache_bytes counts stored elements including batch and layers. qa_f1 uses
multiset overlap. chrf teaches character n-grams with an explicit effective-order
convention, not a claim of identical defaults to any benchmark implementation.
route selects distinct experts deterministically (lower index wins ties), then
normalizes either educational biased logits or V3's unbiased sigmoid affinities.
absorption_trace proves equality within one low-rank attention head. This does
not assert that low-rank MLA equals an independently trained dense attention.
"""
from collections import Counter
import math
import numpy as np


def cache_bytes(batch, layers, tokens, width, bytes_per_element=2):
    """width is 2*n_kv_heads*d_head for K/V, or d_latent+d_rope for MLA."""
    values=(batch,layers,tokens,width,bytes_per_element)
    if any(not isinstance(v,int) or v<0 for v in values):
        raise ValueError('nonnegative integer dimensions required')
    if bytes_per_element == 0: raise ValueError('element width must be positive')
    return math.prod(values)


def qa_f1(prediction_tokens, reference_tokens):
    """Input tokens must already follow the chosen dataset's normalization."""
    if not prediction_tokens or not reference_tokens:
        return float(prediction_tokens == reference_tokens)
    overlap=sum((Counter(prediction_tokens)&Counter(reference_tokens)).values())
    return 2*overlap/(len(prediction_tokens)+len(reference_tokens))


def chrf(hypothesis, reference, max_order=6, beta=2):
    """Remove whitespace; average per-order precisions and recalls; 0..100.

    Only orders present on both sides are used. For official reporting use the
    benchmark implementation, freeze its version, and record its metric signature.
    """
    if max_order<1 or beta<=0: raise ValueError('positive order and beta required')
    h=''.join(hypothesis.split()); r=''.join(reference.split())
    if not h or not r: return 100.0 if h==r else 0.0
    ps=[];rs=[]
    for n in range(1,max_order+1):
        if len(h)<n or len(r)<n: continue
        hc=Counter(h[i:i+n] for i in range(len(h)-n+1))
        rc=Counter(r[i:i+n] for i in range(len(r)-n+1))
        overlap=sum((hc&rc).values())
        ps.append(overlap/sum(hc.values()));rs.append(overlap/sum(rc.values()))
    p=sum(ps)/len(ps);q=sum(rs)/len(rs)
    return 0.0 if p+q==0 else 100*(1+beta**2)*p*q/(beta**2*p+q)


def softmax(values):
    z=np.asarray(values,dtype=float)
    if not np.isfinite(z).all(): raise ValueError('finite logits required')
    e=np.exp(z-np.max(z,axis=-1,keepdims=True))
    return e/e.sum(axis=-1,keepdims=True)


def route(logits,bias,k,mode='book'):
    """Returns zero-based selected indices, selected gates, and integer loads."""
    z=np.asarray(logits,dtype=float);b=np.asarray(bias,dtype=float)
    if z.ndim!=2 or b.shape!=(z.shape[1],) or not 1<=k<=z.shape[1]:
        raise ValueError('router dimensions or top-k are invalid')
    if not np.isfinite(z).all() or not np.isfinite(b).all():raise ValueError('finite router values required')
    if mode=='book': scores=z+b; affinity=z+b
    elif mode=='v3':
        affinity=np.exp(-np.logaddexp(0,-z));scores=affinity+b
    else:raise ValueError('mode must be book or v3')
    idx=np.argsort(-scores,axis=-1,kind='stable')[:,:k]
    chosen=np.take_along_axis(affinity,idx,axis=1)
    gates=softmax(chosen) if mode=='book' else chosen/chosen.sum(axis=1,keepdims=True)
    return idx,gates,np.bincount(idx.ravel(),minlength=z.shape[1])


def bias_update(bias,loads,rate=.01,mode='book'):
    """Book: bounded tanh relative violation; V3: sign step on whole-batch load.

    Call once after collecting the actual forward assignments across relevant
    microbatches/ranks. Recomputing after optimizer updates can change routing.
    An empty batch leaves bias unchanged.
    """
    b=np.asarray(bias,dtype=float);c=np.asarray(loads,dtype=float)
    if b.shape!=c.shape or c.ndim!=1 or len(c)==0 or np.any(c<0):raise ValueError('invalid load vector')
    avg=c.mean()
    if avg==0:return b.copy()
    delta=np.tanh((avg-c)/(avg+1e-6)) if mode=='book' else np.sign(avg-c)
    return b+rate*delta

"""Deterministic CPU teaching reconstruction; NumPy only, no model downloads.

Functions expose the actual mechanics of the inherited MiniDeepSeek lessons.
This is not a reproduction of a 130M-parameter training run or production V3.

from dataclasses import dataclass
import math

import numpy as np

def softmax(x):
    """Normalize the final axis after subtracting its maximum."""
    x = np.asarray(x, dtype=float)
    y = np.exp(x - np.max(x, axis=-1, keepdims=True))
    return y / np.sum(y, axis=-1, keepdims=True)

def ce(logits, targets):
    """Mean next-token negative log likelihood over all supplied positions."""
    z = np.asarray(logits, dtype=float)
    t = np.asarray(targets, dtype=int)
    if z.shape[:-1] != t.shape or t.size == 0:
        raise ValueError('nonempty logits and target positions must align')
    if np.any(t < 0) or np.any(t >= z.shape[-1]):
        raise ValueError('target outside vocabulary')
    a = z - z.max(axis=-1, keepdims=True)
    lp = a - np.log(np.exp(a).sum(axis=-1, keepdims=True))
    return float(-np.take_along_axis(lp, t[..., None], -1).mean())

def pack(stories, eot, vocab):
    """Append one EOT per already-tokenized document; prevent uint16 overflow."""
    ids = [i for story in stories for i in [*story, eot]]
    if not 0 < vocab <= 65536 or any(not 0 <= i < vocab for i in ids):
        raise ValueError('vocabulary/token cannot be represented safely')
    return np.asarray(ids, dtype=np.uint16)

def batch(stream, starts, length):
    """Extract X[i:i+T] and Y[i+1:i+T+1], expanding storage IDs to int64."""
    if length < 1 or not starts or any(i < 0 or i + length >= len(stream) for i in starts):
        raise ValueError('each start needs T+1 tokens')
    return tuple(np.stack([stream[i+shift:i+shift+length] for i in starts]).astype(np.int64)
                 for shift in (0, 1))

def rope(x, offset=0, max_length=32):

    x = np.asarray(x, dtype=float)

import math
import numpy as np

def mtp_alignment(tokens, depth):
    if depth < 1: raise ValueError('depth counts additional tokens beyond main NTP')
    n=max(0,len(tokens)-depth-1)
    return [{'base':i,'conditioning':i+depth,'target':i+depth+1,'token':tokens[i+depth+1]} for i in range(n)]

def rmsnorm(x, eps=1e-6):
    x=np.asarray(x,dtype=float)
    return x/np.sqrt(np.mean(x*x,axis=-1,keepdims=True)+eps)

def mtp_module(h, embedding, projection):
    """Projection+nonlinear toy refinement; not a production Transformer block."""
    u=np.concatenate([rmsnorm(h),rmsnorm(embedding)],axis=-1)
    return np.tanh(u @ np.asarray(projection))

def chain_loss(params,h=.7,target=1.,weight=.1):
    """Scalar two-stage surrogate illustrates all gradient paths."""
    a,b=params;h1=a*h;h2=b*h1
    return .5*(h-1.)**2+weight/2*(.5*(h1-target)**2+.5*(h2-target)**2)

def chain_grad(params,h=.7,target=1.,weight=.1):
    a,b=params;h1=a*h;h2=b*h1;c=weight/2
    return np.array([c*((h1-target)*h+(h2-target)*b*h),c*(h2-target)*h1])

def speculative(p,q):
    p=np.asarray(p,float);q=np.asarray(q,float)
    if np.any(p<0) or np.any(q<0) or not np.isclose(p.sum(),1) or not np.isclose(q.sum(),1):raise ValueError('probability distributions required')
    acceptance=np.divide(p,q,out=np.ones_like(p),where=q>0).clip(0,1)
    accepted=q*acceptance; rejection=1-accepted.sum()
    residual=np.maximum(p-q,0)
    residual=residual/residual.sum() if residual.sum()>0 else np.zeros_like(p)
    return {'acceptance':acceptance,'rejection':float(rejection),'residual':residual,'output':accepted+rejection*residual}

def fp8_values(kind='e4m3fn'):
    """Returns (finite value, unsigned magnitude code), including subnormals."""
    if kind not in ('e4m3fn','e5m2'):raise ValueError(kind)
    m,e,bias=(3,4,7) if kind=='e4m3fn' else (2,5,15)
    out=[]
    for exponent in range(2**e):
      for mantissa in range(2**m):
        if kind=='e4m3fn' and exponent==15 and mantissa==7:continue
        if kind=='e5m2' and exponent==31:continue
        value=mantissa/2**m*2**(1-bias) if exponent==0 else (1+mantissa/2**m)*2**(exponent-bias)
        out.append((value,(exponent<<m)|mantissa))
    return out

def cast_fp8(x,kind='e4m3fn'):
    """Saturating finite RTNE simulation; explicitly excludes NaN/infinity input."""
    if not math.isfinite(float(x)):raise ValueError('finite input required')
    v=abs(float(x)); vals=fp8_values(kind)
    nearest=min(vals,key=lambda z:(abs(v-z[0]),z[1]%2,z[1]))[0]
    return math.copysign(nearest,float(x))

def quantize(x,kind='e4m3fn',amax=None):
    x=np.asarray(x,float)
    maximum=max(v for v,c in fp8_values(kind))
    observed=float(np.max(np.abs(x))) if x.size else 0.
    calibration=observed if amax is None else float(amax)
    if calibration<0:raise ValueError('nonnegative amax required')
    scale=maximum/calibration if calibration>0 else 1.
    q=np.array([cast_fp8(v*scale,kind) for v in x.flat]).reshape(x.shape)
    return {'q':q,'scale':scale,'reconstructed':q/scale,'amax':observed}

def block_linear(x,w,block=128):
    x=np.asarray(x,float);w=np.asarray(w,float)
    if x.ndim!=2 or w.ndim!=2 or x.shape[1]!=w.shape[0] or block<1:raise ValueError('X rows times W input/output columns')
    out=np.zeros((x.shape[0],w.shape[1]),float)

        for i,a in enumerate(qx):out[i,j:jw]+=(a['q']@qw['q'])/(a['scale']*qw['scale'])
    return out

def linear_backward(x,w,dy):

def weighted_gradient(gradients,sizes):
    if len(gradients)!=len(sizes) or not sizes or min(sizes)<=0:raise ValueError('positive sample counts required')
    return sum(np.asarray(g,float)*n for g,n in zip(gradients,sizes))/sum(sizes)

"""Small, explained CPU mathematics; no model downloads or training required.

Run python companion.py for a complete group-relative policy update and a
tool-token mask demonstration. Natural logarithms give losses in nats.
"""
import math

def mean(values):
    """Arithmetic mean; an empty batch is a configuration error."""
    if not values: raise ValueError('empty batch')
    return math.fsum(values)/len(values)

def advantages(rewards, ddof=1, eps=1e-4):
    """Center a same-prompt group, explicitly selecting sample/population std.
    Singleton sample-std groups are rejected, not quietly treated as trainable.
    Equal-reward groups return zeros; KL/other losses may still update weights.
    """
    n=len(rewards)
    if n<=ddof or ddof not in (0,1) or eps<0: raise ValueError('invalid group')
    mu=mean(rewards)
    sd=math.sqrt(math.fsum((r-mu)**2 for r in rewards)/(n-ddof))
    return [0.0]*n if sd==0 else [(r-mu)/(sd+eps) for r in rewards]

def sequence_logprob(probabilities):
    """Sum conditional token log probabilities, including generated EOS."""
    if any(p<=0 or p>1 for p in probabilities): raise ValueError('probability')
    return math.fsum(math.log(p) for p in probabilities)

def clip_term(ratio, advantage, epsilon=.2):
    """Maximized PPO surrogate. Min is essential for negative advantages.
    Clipping limits the incentive, not an enforceable bound on updated policy.
    """
    if ratio<0 or epsilon<0: raise ValueError('ratio/clip')
    clipped=max(1-epsilon,min(1+epsilon,ratio))
    return min(ratio*advantage,clipped*advantage)

def kl(current, reference):
    """Exact forward KL(current||reference), over complete token support."""
    if len(current)!=len(reference) or not math.isclose(sum(current),1) or not math.isclose(sum(reference),1): raise ValueError('distribution')
    if any(x<0 for x in current+reference): raise ValueError('negative mass')
    if any(p>0 and q==0 for p,q in zip(current,reference)): return math.inf
    return math.fsum(p*math.log(p/q) for p,q in zip(current,reference) if p)

def k3(log_current, log_reference):
    """Nonnegative sampled estimator exp(delta)-delta-1, delta=logref-logp.
    Expectation equals forward KL only with samples from current policy;
    old-policy sampling needs explicit importance correction for that identity.
    """
    delta=log_reference-log_current
    return math.expm1(delta)-delta

def token_grpo(ratios, rewards, mask, epsilon=.2, ddof=1):
    """Mean over eligible tokens per response, then mean over responses.
    Equal response weighting differs from one global mean over all tokens.
    """
    a=advantages(rewards,ddof)
    if not (len(ratios)==len(mask)==len(a)): raise ValueError('batch shape')
    rows=[]
    for rs,ms,ai in zip(ratios,mask,a):
        if len(rs)!=len(ms) or sum(ms)<=0: raise ValueError('token mask')
        rows.append(math.fsum(clip_term(r,ai,epsilon)*m for r,m in zip(rs,ms))/sum(ms))
    return -mean(rows)

def entropy(p):
    """Discrete entropy with 0 log 0 = 0."""
    if any(x<0 for x in p) or not math.isclose(sum(p),1): raise ValueError('distribution')
    return -math.fsum(x*math.log(x) for x in p if x)

def softmax(z):
    """Max-subtraction prevents overflow in finite logits."""
    m=max(z); es=[math.exp(x-m) for x in z]; den=math.fsum(es)
    return [x/den for x in es]

def ce_from_logits(z,target):
    """Stable hard-target cross-entropy, not teacher-logit distillation."""
    m=max(z)
    return m+math.log(math.fsum(math.exp(x-m) for x in z))-z[target]

def response_mask(total,prompt):
    """Shifted input t[:-1], targets t[1:]; first response logit P-1.
    A prompt needs at least one token (e.g. BOS); padding is masked separately.
    """
    if not 1<=prompt<total: raise ValueError('prompt/sequence')
    return [int(j>=prompt-1) for j in range(total-1)]

def pass_at_k(p,k):
    """Illustrative iid Bernoulli probability; not finite-sample estimator."""
    if not 0<=p<=1 or k<1: raise ValueError('p/k')
    return 1-(1-p)**k

def empirical_pass_at_k(n,c,k):
    """Unbiased combinatorial estimator from n independent candidates."""
    if not 0<=c<=n or not 1<=k<=n: raise ValueError('counts')
    return 1 if n-c<k else 1-math.comb(n-c,k)/math.comb(n,k)

def votes(answers,correct):
    """Pass event and plurality event, with explicit lexicographic tie break."""
    if not answers: raise ValueError('empty')
    counts={x:answers.count(x) for x in set(answers)}
    winner=min(counts,key=lambda x:(-counts[x],str(x)))
    return int(correct in answers),int(winner==correct)

def gae(rewards,values,gamma=1,lam=.95):
    """Backward generalized advantage recursion, terminal value appended.
    Time indexes policy decisions; interpreter output is an observation.
    """
    if len(values)!=len(rewards)+1: raise ValueError('values include terminal')
    out=[0.0]*len(rewards); acc=0
    for t in range(len(rewards)-1,-1,-1):
        delta=rewards[t]+gamma*values[t+1]-values[t]
        acc=delta+gamma*lam*acc;out[t]=acc
    return out

def miniature_policy_update(logits,rewards,lr=.1):
    """Exact two-action bandit REINFORCE gradient; an integrated RL analogue.
    Enumerates actions, so no sampling noise; reward is detached constant.
    """
    p=softmax(logits); baseline=math.fsum(pi*r for pi,r in zip(p,rewards))
    # d E[R]/d z_j = p_j (R_j-E[R]); gradient ascent.
    gradient=[pi*(r-baseline) for pi,r in zip(p,rewards)]
    updated=[z+lr*g for z,g in zip(logits,gradient)]
    return dict(before=p,expected_reward=baseline,gradient=gradient,logits=updated,after=softmax(updated))
